# ============================================================
#  信息传输助手 —— 配置说明
# ============================================================
#
#  怎么用：
#    1. 把这个文件里需要填的地方填好
#    2. 保存（编码保持 UTF-8）
#    3. 双击同目录的「生成传输助手.bat」
#    4. 会生成「传输助手.bat」，把它发到目标电脑双击即可
#
#  说明：
#    - 以 # 开头的行是注释，不用管
#    - 等号两边可以有空格，也可以没有
#    - 只填「服务器地址」和「设备密钥」两项就能用
#
# ============================================================


# ------------------------------------------------------------
#  服务器地址（必填）
# ------------------------------------------------------------
#  填你服务器的公网 IP 或域名就行，不用写 http://
#
#  正确写法举例：
#      203.0.113.10
#      123.45.67.89
#      relay.example.com
#      http://203.0.113.10:8080      （写了 http:// 和端口也行）
#
服务器地址=


# ------------------------------------------------------------
#  端口（可选，默认 8080）
# ------------------------------------------------------------
#  如果上面的地址里已经写了端口，这里会被忽略
#
端口=8080


# ------------------------------------------------------------
#  设备密钥（必填）
# ------------------------------------------------------------
#  这个密钥来自你的服务器，在 /opt/server/config.json 里的 device_key
#  或者你运行 setup.py 生成配置时屏幕上显示的那串
#
#  它是一串 32 位的字母数字，用来防止别人随便连你的服务器
#
设备密钥=


# ------------------------------------------------------------
#  传输方式（可选，默认 1）
# ------------------------------------------------------------
#  1 = 仅 OSS          数据经阿里云 OSS 中转，兼容性最好，任何网络都能用
#  2 = IPv6 优先       双方都有公网 IPv6 时直连（最快、且不花 OSS 流量），
#                      任一方没有 IPv6 就自动回落 OSS
#  3 = 仅 IPv6         只用 IPv6，绝不回落；没有 IPv6 时无法传输
#
#  不确定就填 1。IPv6 需要「目标电脑」和「你访问用的设备」都有公网 IPv6。
#
传输方式=1


# ------------------------------------------------------------
#  开机自启（可选，默认 1）
# ------------------------------------------------------------
#  1 = 目标电脑开机后自动在后台运行（推荐）
#  0 = 只在双击时运行
#
开机自启=1
